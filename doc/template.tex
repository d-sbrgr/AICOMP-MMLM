% Fix for xltabular/longtable table counter double-increment issue
\usepackage{etoolbox}
\makeatletter
% When using \captionof{table} before xltabular, the counter is already incremented
% We need to decrement it before xltabular increments it again
\AtBeginEnvironment{xltabular}{\addtocounter{table}{-1}}
% Also patch longtable's \LT@array to prevent it from incrementing when \caption is used inside
\patchcmd{\LT@array}{\refstepcounter{table}}{}{}{}
\makeatother

  \vspace{-16pt}

}
