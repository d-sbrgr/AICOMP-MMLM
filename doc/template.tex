% Redefine subfigure to work with pandoc-crossref syntax and add spacing
% pandoc-crossref uses: \begin{subfigure}{width}

  \origsubfigure[t]{#1}%
  \centering%
}{%

  \hspace{1em}% Add fixed 1em space after each subfigure
}
